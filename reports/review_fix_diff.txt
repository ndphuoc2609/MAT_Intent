+def totals(path=DEFAULT_PATH):
+    """Trả (calls, cost_known, unknown_count, entries). File không có -> 0."""
+    calls, cost, unknown, n = 0, 0.0, 0, 0
+    if not os.path.exists(path):
+        return calls, cost, unknown, n
+    for line in open(path):
+        if not line.strip(): continue
+        try: rec = json.loads(line)
+        except json.JSONDecodeError: continue
+        n += 1
+        calls += int(rec.get("attempts", 0) or 0)
+        if rec.get("cost_usd_est") is None:
+            unknown += 1
+        else:
+            cost += float(rec.get("cost_usd_est") or 0)
+    return calls, cost, unknown, n

+def record(path, *, run_id, kind, attempts, cost_usd_est, model="", note=""):

+    os.makedirs(os.path.dirname(path), exist_ok=True)

+           "kind": kind, "attempts": int(attempts),
+           "cost_usd_est": cost_usd_est,
+           "cost_unknown": cost_usd_est is None,
+           "model": model, "note": note}
+    with open(path, "a") as f:

+def guard(path=DEFAULT_PATH, *, max_calls=300, max_cost=2.00, est_attempts=1,
+          est_cost=0.0, cost_unknown=False):

+    không xác minh được -> chặn spend (fail-closed)."""

+    if cost + est_cost > max_cost:

+def build_request_map(old_requests, new_requests):
+    """Map old request id -> new request id theo logical key. Key trùng nhau

+    """Map old pending id -> new pending id theo logical key."""

+# ---------------- R03 ----------------
+class LedgerSender(FakeSender):
+    """Sender mô phỏng receiver ledger bền vững RIÊNG (sống qua restart, commit
+    ngay khi effect xảy ra). crash=True: effect đã ghi nhưng raise trước khi
+    engine lưu kết quả -> engine rollback toàn bộ transaction."""
+    def __init__(self, ledger, crash=False):
+        super().__init__()
+        self.ledger = ledger
+        self.crash = crash
+    def send(self, payload):
+        self.ledger.append(payload["dedupe_key"])  # effect bền vững, không dedupe (baseline)
+        if self.crash:
+            raise RuntimeError("crash after receiver effect, before result persisted")
+        return {"status": "sent_to_fake_ledger", "id": payload["dedupe_key"]}
+

+def _load_yaml(path):
+    # Parser YAML tối giản cho file policy (không cần PyYAML): đọc list/dict
+    # nông với key: value và list "- item".
+    out, cur = {}, None
+    for raw in open(path):
+        line = raw.rstrip("\n")
+        if not line.strip() or line.lstrip().startswith("#"):
+            continue
+        if line.startswith("  - "):
+            out.setdefault(cur, []).append(line[4:].strip())
+            continue
+        if ":" in line and not line.startswith(" "):
+            key, val = line.split(":", 1)
+            cur = key.strip()
+            val = val.split("#")[0].strip()
+            if val:
+                try: val = int(val)
+                except ValueError: pass
+                out[cur] = val
+            else:
+                out.setdefault(cur, [])
+    return out
+

+    def test_request_and_pending_remap(self):
+        old_reqs = [_req("old-r1", "price", "OPEN", {"vehicle": {"value": "Creta"}})]
+        new_reqs = [_req("new-r9", "price", "OPEN", {"vehicle": {"value": "Creta"}})]

+                  "expected_slot": None, "options": ["yes", "no"], "status": "OPEN"}]
+        pend_map = build_pending_map(old_p, new_p, old_reqs, new_reqs)
+        self.assertEqual(pend_map, {"old-p1": "new-p7"})
